খুঁজুন এই দুটো লাইন (runStepSmart ফাংশনের ভিতরে):

            const result = await runStepOnce(stepName, fnFactory);

মাঝখানে ১ লাইন ঢুকিয়ে এমন করুন:

            try { rjRotationFeedback(result.win); } catch(e) {}

【 প্যাচ ৩ 】 Rotation module  (পুরো ব্লক যোগ)
────────────────────────────────────────────────────────────

খুঁজুন এই লাইনটা:

    // ==================== UPLOAD TAB ====================

ঠিক তার আগে নিচের পুরো ব্লকটা পেস্ট করুন:

    // ==================== PROXY ROTATION (additive — বাকি কিছু বদলায় না) ====================
    // rotation toggle (Fetch tab-এর "Proxy") ON থাকলে: saved proxy list = pool.
    // success হলে current IP ধরে রাখে (sticky), error হলে এইমাত্র fail করা IP বাদ দিয়ে নতুন random তোলে।
    const rjRot = { current: null, lastFailedId: null };
    function rjRotationEnabled() {
        return !!document.getElementById('ivac-parallel-proxy-rotation-toggle')?.classList.contains('on');
    }
    function rjRotationPool() {
        try { return (typeof loadProxies === 'function') ? loadProxies() : []; } catch(e) { return []; }
    }
    function rjPickRandomProxy() {
        const pool = rjRotationPool();
        if (!pool.length) return null;
        let choices = pool;
        if (pool.length > 1 && rjRot.lastFailedId) {
            const filtered = pool.filter(p => p.id !== rjRot.lastFailedId);
            if (filtered.length) choices = filtered;   // এইমাত্র fail করা IP বাদ
        }
        return choices[Math.floor(Math.random() * choices.length)];
    }
    // gmFetch এটা দিয়েই ঠিক করে কোন proxy দিয়ে যাবে
    function rjPickProxyForRequest(isConnectionCheck) {
        if (isConnectionCheck) return null;                 // keep-alive HEAD সবসময় direct
        if (rjRotationEnabled()) {
            const pool = rjRotationPool();
            if (!pool.length) return (typeof window !== 'undefined') ? (window._rjActiveProxy || null) : null; // pool খালি → পুরোনো আচরণ
            if (!rjRot.current) {
                rjRot.current = rjPickRandomProxy();
                if (rjRot.current) logStatus(`🔀 Proxy → ${rjRot.current.label}`, 'y');
            }
            return rjRot.current || null;
        }
        return (typeof window !== 'undefined') ? (window._rjActiveProxy || null) : null;
    }
    // runStepSmart প্রতি চেষ্টার পর এটা ডাকে: success=ধরে রাখো, fail=ঘোরাও
    function rjRotationFeedback(win) {
        if (!rjRotationEnabled()) return;
        if (win) { rjRot.lastFailedId = null; }             // sticky: current যেমন আছে থাকুক
        else { if (rjRot.current) rjRot.lastFailedId = rjRot.current.id; rjRot.current = null; } // পরের বার নতুন random
    }
